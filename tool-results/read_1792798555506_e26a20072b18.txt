    62→function useSettings() {
    63→  const { get } = useAdminApi();
    64→  const [settings, setSettings] = useState<SiteSettings | null>(null);
    65→  const [integrations, setIntegrations] = useState<Integrations | null>(null);
    66→  const [loading, setLoading] = useState(true);
    67→
    68→  const load = useCallback(async () => {
    69→    const res = await get<{ settings: SiteSettings; integrations: Integrations }>("/api/admin/settings");
    70→    if (res) {
    71→      setSettings(res.settings);
    72→      setIntegrations(res.integrations);
    73→    }
    74→    setLoading(false);
    75→  }, [get]);
    76→
    77→  useEffect(() => {
    78→    let cancelled = false;
    79→    (async () => {
    80→      const res = await get<{ settings: SiteSettings; integrations: Integrations }>("/api/admin/settings");
    81→      if (!cancelled) {
    82→        if (res) {
    83→          setSettings(res.settings);
    84→          setIntegrations(res.integrations);
    85→        }
    86→        setLoading(false);
    87→      }
    88→    })();
    89→    return () => {
    90→      cancelled = true;
    91→    };
    92→  }, [get]);
    93→
    94→  return { settings, integrations, loading, load };
    95→}

   496→function OutboxTab() {
   497→  const { get, send } = useAdminApi();
   498→  const [items, setItems] = useState<OutboxDTO[] | null>(null);
   499→  const [loading, setLoading] = useState(true);
   500→  const [draining, setDraining] = useState(false);
   501→
   502→  const load = useCallback(async () => {
   503→    const res = await get<{ health: { pending: number; failed: number; sent: number }; items: OutboxDTO[] }>("/api/admin/notifications");
   504→    if (res) setItems(res.items);
   505→    setLoading(false);
   506→  }, [get]);
   507→
   508→  useEffect(() => {
   509→    let cancelled = false;
   510→    (async () => {
   511→      const res = await get<{ health: { pending: number; failed: number; sent: number }; items: OutboxDTO[] }>("/api/admin/notifications");
   512→      if (!cancelled) {
   513→        if (res) setItems(res.items);
   514→        setLoading(false);
   515→      }
   516→    })();
   517→    return () => {
   518→      cancelled = true;
   519→    };
   520→  }, [get]);
   521→
   522→  async function drain() {
   523→    setDraining(true);
   524→    const res = await send<{ success: boolean; sent?: number; failed?: number; message?: string }>("/api/admin/settings", "POST");
   525→    setDraining(false);
   526→    if (res) {
   527→      if (res.success) toast.success(`Outbox drained — ${res.sent ?? 0} sent, ${res.failed ?? 0} failed.`);
   528→      else toast.info(res.message ?? "Nothing to send.");
   529→      await load();
   530→    }
   531→  }
   532→
   533→  return (
   534→    <div className="space-y-4">
   535→      <div className="flex flex-wrap items-center justify-between gap-3">
   536→        <p className="text-sm text-muted-foreground">Every customer email is queued here first — ordering never blocks on delivery.</p>
   537→        <Button onClick={() => void drain()} disabled={draining} className="bg-forest text-cream hover:bg-[#23503E]">
   538→          <Send className="h-4 w-4" /> {draining ? "Sending…" : "Drain outbox now"}
   539→        </Button>
   540→      </div>
   541→      <div className="rounded-2xl border border-border bg-card shadow-sm">
   542→        {loading && !items ? (
   543→          <div className="p-4"><SkeletonRows rows={5} /></div>
   544→        ) : (items ?? []).length === 0 ? (
   545→          <p className="p-6 text-center text-sm text-muted-foreground">Outbox is empty.</p>
   546→        ) : (
   547→          <div className="max-h-[62vh] overflow-auto">
   548→            <Table className="min-w-[720px]">
   549→              <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur">
   550→                <TableRow>
   551→                  <TableHead>Subject</TableHead>
   552→                  <TableHead>Type</TableHead>
   553→                  <TableHead>Recipient</TableHead>
   554→                  <TableHead>Status</TableHead>
   555→                  <TableHead className="text-center">Attempts</TableHead>
   556→                  <TableHead>Created</TableHead>
   557→                </TableRow>
   558→              </TableHeader>
   559→              <TableBody>
   560→                {(items ?? []).map((i) => (
   561→                  <TableRow key={i.id}>
   562→                    <TableCell className="max-w-[260px] truncate font-medium">{i.subject}</TableCell>
   563→                    <TableCell><Badge variant="outline" className="text-[11px]">{i.type}</Badge></TableCell>
   564→                    <TableCell className="text-muted-foreground">{i.recipient ?? "—"}</TableCell>
   565→                    <TableCell><OutboxStatusBadge status={i.status} /></TableCell>
   566→                    <TableCell className="text-center">{i.attempts}</TableCell>
   567→                    <TableCell className="whitespace-nowrap text-xs text-muted-foreground">{formatDate(i.createdAt)}</TableCell>
   568→                  </TableRow>
   569→                ))}
   570→              </TableBody>
   571→            </Table>
   572→          </div>
   573→        )}
   574→      </div>
   575→    </div>
   576→  );
   577→}

   581→function AuditTab() {
   582→  const { get } = useAdminApi();
   583→  const [events, setEvents] = useState<AuditDTO[] | null>(null);
   584→  const [total, setTotal] = useState(0);
   585→  const [page, setPage] = useState(1);
   586→  const [actionFilter, setActionFilter] = useState("all");
   587→  const [searchDraft, setSearchDraft] = useState("");
   588→  const [loading, setLoading] = useState(true);
   589→  const PAGE_SIZE = 25;
   590→
   591→  useEffect(() => {
   592→    let cancelled = false;
   593→    (async () => {
   594→      const params = new URLSearchParams();
   595→      params.set("page", String(page));
   596→      params.set("pageSize", String(PAGE_SIZE));
   597→      if (actionFilter !== "all") params.set("action", actionFilter);
   598→      const res = await get<{ total: number; events: AuditDTO[] }>(`/api/admin/audit?${params.toString()}`);
   599→      if (!cancelled) {
   600→        if (res) {
   601→          setEvents(res.events);
   602→          setTotal(res.total);
   603→        }
   604→        setLoading(false);
   605→      }
   606→    })();
   607→    return () => {
   608→      cancelled = true;
   609→    };
   610→  }, [get, page, actionFilter]);
   611→
   612→  useEffect(() => {
   613→    const t = setTimeout(() => setActionFilter(searchDraft.trim() || "all"), 350);
   614→    return () => clearTimeout(t);
   615→  }, [searchDraft]);
   616→
   617→  const pageCount = Math.max(1, Math.ceil(total / PAGE_SIZE));
   618→
   619→  return (
   620→    <TooltipProvider>
   621→      <div className="space-y-4">
   622→        <div className="flex flex-wrap items-center gap-3">
   623→          <div className="relative flex-1 sm:max-w-xs">
   624→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   625→            <Input value={searchDraft} onChange={(e) => setSearchDraft(e.target.value)} placeholder="Filter by action e.g. payment.approve" className="pl-9" aria-label="Filter audit log" />
   626→          </div>
   627→          <p className="text-sm text-muted-foreground">{total.toLocaleString()} events</p>
   628→        </div>
   629→
   630→        <div className="rounded-2xl border border-border bg-card shadow-sm">
   631→          {loading && !events ? (
   632→            <div className="p-4"><SkeletonRows rows={8} /></div>
   633→          ) : (events ?? []).length === 0 ? (
   634→            <p className="p-6 text-center text-sm text-muted-foreground">No audit events match.</p>
   635→          ) : (
   636→            <div className="max-h-[62vh] overflow-auto">
   637→              <Table className="min-w-[720px]">
   638→                <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur">
   639→                  <TableRow>
   640→                    <TableHead>Action</TableHead>
   641→                    <TableHead>Actor</TableHead>
   642→                    <TableHead>Target</TableHead>
   643→                    <TableHead>When</TableHead>
   644→                  </TableRow>
   645→                </TableHeader>
   646→                <TableBody>
   647→                  {(events ?? []).map((e) => (
   648→                    <TableRow key={e.id}>
   649→                      <TableCell>
   650→                        <Tooltip>
   651→                          <TooltipTrigger asChild>
   652→                            <Badge variant="outline" className="font-mono text-[11px]">{e.action}</Badge>
   653→                          </TooltipTrigger>
   654→                          {Object.keys(e.meta ?? {}).length > 0 ? (
   655→                            <TooltipContent className="max-w-72 break-all text-xs">
   656→                              <pre className="whitespace-pre-wrap font-mono">{JSON.stringify(e.meta, null, 1)}</pre>
   657→                            </TooltipContent>
   658→                          ) : null}
   659→                        </Tooltip>
   660→                      </TableCell>
   661→                      <TableCell>{e.actor}</TableCell>
   662→                      <TableCell className="max-w-[220px] truncate text-muted-foreground">{e.target ?? "—"}</TableCell>
   663→                      <TableCell className="whitespace-nowrap text-xs text-muted-foreground">{formatDate(e.createdAt)}</TableCell>
   664→                    </TableRow>
   665→                  ))}
   666→                </TableBody>
   667→              </Table>
   668→            </div>
   669→          )}
   670→          <div className="flex items-center justify-between border-t border-border px-4 py-3 text-sm">
   671→            <span className="text-muted-foreground">Page {page} of {pageCount}</span>
   672→            <div className="flex gap-2">
   673→              <Button variant="outline" size="sm" disabled={page <= 1 || loading} onClick={() => setPage((p) => Math.max(1, p - 1))}>Prev</Button>
   674→              <Button variant="outline" size="sm" disabled={page >= pageCount || loading} onClick={() => setPage((p) => p + 1)}>Next</Button>
   675→            </div>
   676→          </div>
   677→        </div>
   678→      </div>
   679→    </TooltipProvider>
   680→  );
   681→}
