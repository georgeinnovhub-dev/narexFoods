   206→   206→function CouponDialog({
   207→   207→  open,
   208→   208→  coupon,
   209→   209→  onOpenChange,
   210→   210→  onSaved,
   211→   211→}: {
   212→   212→  open: boolean;
   213→   213→  coupon: CouponDTO | null;
   214→   214→  onOpenChange: (open: boolean) => void;
   215→   215→  onSaved: () => Promise<void>;
   216→   216→}) {
   217→   217→  const { send } = useAdminApi();
   218→   218→  const [code, setCode] = useState("");
   219→   219→  const [type, setType] = useState<"percent" | "fixed">("percent");
   220→   220→  const [percent, setPercent] = useState(10);
   221→   221→  const [fixed, setFixed] = useState<number | null>(500);
   222→   222→  const [minSubtotal, setMinSubtotal] = useState<number | null>(0);
   223→   223→  const [startsAt, setStartsAt] = useState("");
   224→   224→  const [endsAt, setEndsAt] = useState("");
   225→   225→  const [maxRedemptions, setMaxRedemptions] = useState("");
   226→   226→  const [active, setActive] = useState(true);
   227→   227→  const [saving, setSaving] = useState(false);
   228→   228→
   229→   229→  // Re-initialize the form whenever the editor opens for a different coupon.
   230→   230→  const [initKey, setInitKey] = useState<string | null>(null);
   231→   231→  const initKeyNow = open ? (coupon?.id ?? "new") : null;
   232→   232→  if (initKeyNow !== initKey) {
   233→   233→    setInitKey(initKeyNow);
   234→   234→    if (initKeyNow !== null) {
   235→   235→      setCode(coupon?.code ?? "");
   236→   236→      setType((coupon?.type as "percent" | "fixed") ?? "percent");
   237→   237→      setPercent(coupon?.type === "percent" ? coupon.value : 10);
   238→   238→      setFixed(coupon?.type === "fixed" ? coupon.value : 500);
   239→   239→      setMinSubtotal(coupon?.minSubtotal ?? 0);
   240→   240→      setStartsAt(coupon?.startsAt ? toLocalInput(coupon.startsAt) : "");
   241→   241→      setEndsAt(coupon?.endsAt ? toLocalInput(coupon.endsAt) : "");
   242→   242→      setMaxRedemptions(coupon?.maxRedemptions ? String(coupon.maxRedemptions) : "");
   243→   243→      setActive(coupon?.active ?? true);
   244→   244→    }
   245→   245→  }
   246→   246→
   247→   247→  function toIso(local: string): string | undefined {
   248→   248→    if (!local) return undefined;
   249→   249→    const d = new Date(local);
   250→   250→    return Number.isNaN(d.getTime()) ? undefined : d.toISOString();
   251→   251→  }
   252→   252→
   253→   253→  async function save() {
   254→   254→    const cleanCode = code.trim().toUpperCase();
   255→   255→    if (!/^[A-Z0-9]{2,30}$/.test(cleanCode)) return toast.error("Code must be 2–30 letters/numbers (no spaces).");
   256→   256→    const value = type === "percent" ? Math.round(percent) : (fixed ?? 0);
   257→   257→    if (value < 1) return toast.error("Discount value must be at least 1.");
   258→   258→    if (type === "percent" && value > 100) return toast.error("Percentage discount cannot exceed 100.");
   259→   259→    setSaving(true);
   260→   260→    const payload = {
   261→   261→      code: cleanCode,
   262→   262→      type,
   263→   263→      value,
   264→   264→      minSubtotal: minSubtotal ?? 0,
   265→   265→      ...(toIso(startsAt) ? { startsAt: toIso(startsAt) } : {}),
   266→   266→      ...(toIso(endsAt) ? { endsAt: toIso(endsAt) } : {}),
   267→   267→      ...(maxRedemptions.trim() ? { maxRedemptions: Math.max(1, Math.round(Number(maxRedemptions)) || 1) } : {}),
   268→   268→      active,
   269→   269→    };
   270→   270→    const res = await send(coupon ? "/api/admin/coupons" : "/api/admin/coupons", coupon ? "PATCH" : "POST", coupon ? { id: coupon.id, ...payload } : payload);
   271→   271→    setSaving(false);
   272→   272→    if (res !== null) {
   273→   273→      toast.success(coupon ? "Coupon updated." : `Coupon ${cleanCode} created.`);
   274→   274→      onOpenChange(false);
   275→   275→      await onSaved();
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  return (
   280→   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→   281→      <DialogContent className="sm:max-w-md">
   282→   282→        <DialogHeader>
   283→   283→          <DialogTitle className="font-display">{coupon ? `Edit ${coupon.code}` : "New coupon"}</DialogTitle>
   284→   284→          <DialogDescription>{coupon ? "Code, type and window can be changed." : "Percent discounts apply to the food subtotal."}</DialogDescription>
   285→   285→        </DialogHeader>
   286→   286→        <div className="space-y-3">
   287→   287→          <div className="grid gap-3 sm:grid-cols-2">
   288→   288→            <div className="space-y-1.5">
   289→   289→              <Label htmlFor="c-code">Code</Label>
   290→   290→              <Input id="c-code" value={code} onChange={(e) => setCode(e.target.value.toUpperCase())} placeholder="WELCOME10" className="font-mono" />
   291→   291→            </div>
   292→   292→            <div className="space-y-1.5">
   293→   293→              <Label>Type</Label>
   294→   294→              <Select value={type} onValueChange={(v) => setType(v as "percent" | "fixed")}>
   295→   295→                <SelectTrigger><SelectValue /></SelectTrigger>
   296→   296→                <SelectContent>
   297→   297→                  <SelectItem value="percent">Percent off</SelectItem>
   298→   298→                  <SelectItem value="fixed">Fixed amount off</SelectItem>
   299→   299→                </SelectContent>
   300→   300→              </Select>
   301→   301→            </div>
   302→   302→          </div>
   303→   303→          {type === "percent" ? (
   304→   304→            <div className="space-y-1.5">
   305→   305→              <Label htmlFor="c-percent">Discount (%)</Label>
   306→   306→              <Input id="c-percent" type="number" min={1} max={100} value={percent} onChange={(e) => setPercent(Number(e.target.value))} />
   307→   307→            </div>
   308→   308→          ) : (
   309→   309→            <div className="space-y-1.5">
   310→   310→              <Label>Discount amount</Label>
   311→   311→              <PesewasInput value={fixed} onChange={setFixed} />
   312→   312→            </div>
   313→   313→          )}
   314→   314→          <div className="space-y-1.5">
   315→   315→            <Label>Minimum subtotal (0 = none)</Label>
   316→   316→            <PesewasInput value={minSubtotal} onChange={setMinSubtotal} />
   317→   317→          </div>
   318→   318→          <div className="grid gap-3 sm:grid-cols-2">
   319→   319→            <div className="space-y-1.5">
   320→   320→              <Label htmlFor="c-start">Starts</Label>
   321→   321→              <Input id="c-start" type="datetime-local" value={startsAt} onChange={(e) => setStartsAt(e.target.value)} />
   322→   322→            </div>
   323→   323→            <div className="space-y-1.5">
   324→   324→              <Label htmlFor="c-end">Ends</Label>
   325→   325→              <Input id="c-end" type="datetime-local" value={endsAt} onChange={(e) => setEndsAt(e.target.value)} />
   326→   326→            </div>
   327→   327→          </div>
   328→   328→          <div className="flex items-end gap-4">
   329→   329→            <div className="flex-1 space-y-1.5">
   330→   330→              <Label htmlFor="c-max">Max redemptions (blank = ∞)</Label>
   331→   331→              <Input id="c-max" type="number" min={1} value={maxRedemptions} onChange={(e) => setMaxRedemptions(e.target.value)} />
   332→   332→            </div>
   333→   333→            <label className="flex items-center gap-2 pb-2 text-sm">
   334→   334→              <Checkbox checked={active} onCheckedChange={(c) => setActive(c === true)} /> Active
   335→   335→            </label>
   336→   336→          </div>
   337→   337→        </div>
   338→   338→        <DialogFooter>
   339→   339→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
   340→   340→          <Button onClick={() => void save()} disabled={saving} className="bg-forest text-cream hover:bg-[#23503E]">{saving ? "Saving…" : coupon ? "Save changes" : "Create coupon"}</Button>
   341→   341→        </DialogFooter>
   342→   342→      </DialogContent>
   343→   343→    </Dialog>
   344→   344→  );
   345→   345→}

   349→   349→function CampaignsTab() {
   350→   350→  const { get } = useAdminApi();
   351→   351→  const [campaigns, setCampaigns] = useState<CampaignAdminDTO[] | null>(null);
   352→   352→  const [loading, setLoading] = useState(true);
   353→   353→  const [editing, setEditing] = useState<CampaignAdminDTO | "new" | null>(null);
   354→   354→  const [deleting, setDeleting] = useState<CampaignAdminDTO | null>(null);
   355→   355→
   356→   356→  const load = useCallback(async () => {
   357→   357→    const res = await get<{ campaigns: CampaignAdminDTO[] }>("/api/admin/campaigns");
   358→   358→    if (res) setCampaigns(res.campaigns);
   359→   359→    setLoading(false);
   360→   360→  }, [get]);
   361→   361→
   362→   362→  useEffect(() => {
   363→   363→    let cancelled = false;
   364→   364→    (async () => {
   365→   365→      const res = await get<{ campaigns: CampaignAdminDTO[] }>("/api/admin/campaigns");
   366→   366→      if (!cancelled) {
   367→   367→        if (res) setCampaigns(res.campaigns);
   368→   368→        setLoading(false);
   369→   369→      }
   370→   370→    })();
   371→   371→    return () => {
   372→   372→      cancelled = true;
   373→   373→    };
   374→   374→  }, [get]);
   375→   375→
   376→   376→  const totals = (campaigns ?? []).reduce(
   377→   377→    (acc, c) => ({ impressions: acc.impressions + c.impressions, clicks: acc.clicks + c.clicks, dismissals: acc.dismissals + c.dismissals }),
   378→   378→    { impressions: 0, clicks: 0, dismissals: 0 }
   379→   379→  );
   380→   380→  const ctr = totals.impressions > 0 ? ((totals.clicks / totals.impressions) * 100).toFixed(1) : "0.0";
   381→   381→
   382→   382→  async function setStatus(c: CampaignAdminDTO, status: string) {
   383→   383→    const res = await patchCampaign({ id: c.id, status });
   384→   384→    if (res !== null) {
   385→   385→      toast.success(`Campaign “${c.title}” → ${status}.`);
   386→   386→      await load();
   387→   387→    }
   388→   388→  }
   389→   389→
   390→   390→  async function remove() {
   391→   391→    if (!deleting) return;
   392→   392→    try {
   393→   393→      await apiSend(`/api/admin/campaigns?id=${encodeURIComponent(deleting.id)}`, "DELETE");
   394→   394→      toast.success("Campaign deleted.");
   395→   395→      setDeleting(null);
   396→   396→      await load();
   397→   397→    } catch (err) {
   398→   398→      toast.error(errMessage(err));
   399→   399→    }
   400→   400→  }
   401→   401→
   402→   402→  return (
   403→   403→    <div className="space-y-4">
   404→   404→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">

   509→   509→function toLocalInput(iso: string): string {
   510→   510→  const d = new Date(iso);
   511→   511→  if (Number.isNaN(d.getTime())) return "";
   512→   512→  const pad = (n: number) => String(n).padStart(2, "0");
   513→   513→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   514→   514→}
