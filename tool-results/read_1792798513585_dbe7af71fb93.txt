   206→function CouponDialog({
   207→  open,
   208→  coupon,
   209→  onOpenChange,
   210→  onSaved,
   211→}: {
   212→  open: boolean;
   213→  coupon: CouponDTO | null;
   214→  onOpenChange: (open: boolean) => void;
   215→  onSaved: () => Promise<void>;
   216→}) {
   217→  const { send } = useAdminApi();
   218→  const [code, setCode] = useState("");
   219→  const [type, setType] = useState<"percent" | "fixed">("percent");
   220→  const [percent, setPercent] = useState(10);
   221→  const [fixed, setFixed] = useState<number | null>(500);
   222→  const [minSubtotal, setMinSubtotal] = useState<number | null>(0);
   223→  const [startsAt, setStartsAt] = useState("");
   224→  const [endsAt, setEndsAt] = useState("");
   225→  const [maxRedemptions, setMaxRedemptions] = useState("");
   226→  const [active, setActive] = useState(true);
   227→  const [saving, setSaving] = useState(false);
   228→
   229→  // Re-initialize the form whenever the editor opens for a different coupon.
   230→  const [initKey, setInitKey] = useState<string | null>(null);
   231→  const initKeyNow = open ? (coupon?.id ?? "new") : null;
   232→  if (initKeyNow !== initKey) {
   233→    setInitKey(initKeyNow);
   234→    if (initKeyNow !== null) {
   235→      setCode(coupon?.code ?? "");
   236→      setType((coupon?.type as "percent" | "fixed") ?? "percent");
   237→      setPercent(coupon?.type === "percent" ? coupon.value : 10);
   238→      setFixed(coupon?.type === "fixed" ? coupon.value : 500);
   239→      setMinSubtotal(coupon?.minSubtotal ?? 0);
   240→      setStartsAt(coupon?.startsAt ? toLocalInput(coupon.startsAt) : "");
   241→      setEndsAt(coupon?.endsAt ? toLocalInput(coupon.endsAt) : "");
   242→      setMaxRedemptions(coupon?.maxRedemptions ? String(coupon.maxRedemptions) : "");
   243→      setActive(coupon?.active ?? true);
   244→    }
   245→  }
   246→
   247→  function toIso(local: string): string | undefined {
   248→    if (!local) return undefined;
   249→    const d = new Date(local);
   250→    return Number.isNaN(d.getTime()) ? undefined : d.toISOString();
   251→  }
   252→
   253→  async function save() {
   254→    const cleanCode = code.trim().toUpperCase();
   255→    if (!/^[A-Z0-9]{2,30}$/.test(cleanCode)) return toast.error("Code must be 2–30 letters/numbers (no spaces).");
   256→    const value = type === "percent" ? Math.round(percent) : (fixed ?? 0);
   257→    if (value < 1) return toast.error("Discount value must be at least 1.");
   258→    if (type === "percent" && value > 100) return toast.error("Percentage discount cannot exceed 100.");
   259→    setSaving(true);
   260→    const payload = {
   261→      code: cleanCode,
   262→      type,
   263→      value,
   264→      minSubtotal: minSubtotal ?? 0,
   265→      ...(toIso(startsAt) ? { startsAt: toIso(startsAt) } : {}),
   266→      ...(toIso(endsAt) ? { endsAt: toIso(endsAt) } : {}),
   267→      ...(maxRedemptions.trim() ? { maxRedemptions: Math.max(1, Math.round(Number(maxRedemptions)) || 1) } : {}),
   268→      active,
   269→    };
   270→    const res = await send(coupon ? "/api/admin/coupons" : "/api/admin/coupons", coupon ? "PATCH" : "POST", coupon ? { id: coupon.id, ...payload } : payload);
   271→    setSaving(false);
   272→    if (res !== null) {
   273→      toast.success(coupon ? "Coupon updated." : `Coupon ${cleanCode} created.`);
   274→      onOpenChange(false);
   275→      await onSaved();
   276→    }
   277→  }
   278→
   279→  return (
   280→    <Dialog open={open} onOpenChange={onOpenChange}>
   281→      <DialogContent className="sm:max-w-md">
   282→        <DialogHeader>
   283→          <DialogTitle className="font-display">{coupon ? `Edit ${coupon.code}` : "New coupon"}</DialogTitle>
   284→          <DialogDescription>{coupon ? "Code, type and window can be changed." : "Percent discounts apply to the food subtotal."}</DialogDescription>
   285→        </DialogHeader>
   286→        <div className="space-y-3">
   287→          <div className="grid gap-3 sm:grid-cols-2">
   288→            <div className="space-y-1.5">
   289→              <Label htmlFor="c-code">Code</Label>
   290→              <Input id="c-code" value={code} onChange={(e) => setCode(e.target.value.toUpperCase())} placeholder="WELCOME10" className="font-mono" />
   291→            </div>
   292→            <div className="space-y-1.5">
   293→              <Label>Type</Label>
   294→              <Select value={type} onValueChange={(v) => setType(v as "percent" | "fixed")}>
   295→                <SelectTrigger><SelectValue /></SelectTrigger>
   296→                <SelectContent>
   297→                  <SelectItem value="percent">Percent off</SelectItem>
   298→                  <SelectItem value="fixed">Fixed amount off</SelectItem>
   299→                </SelectContent>
   300→              </Select>
   301→            </div>
   302→          </div>
   303→          {type === "percent" ? (
   304→            <div className="space-y-1.5">
   305→              <Label htmlFor="c-percent">Discount (%)</Label>
   306→              <Input id="c-percent" type="number" min={1} max={100} value={percent} onChange={(e) => setPercent(Number(e.target.value))} />
   307→            </div>
   308→          ) : (
   309→            <div className="space-y-1.5">
   310→              <Label>Discount amount</Label>
   311→              <PesewasInput value={fixed} onChange={setFixed} />
   312→            </div>
   313→          )}
   314→          <div className="space-y-1.5">
   315→            <Label>Minimum subtotal (0 = none)</Label>
   316→            <PesewasInput value={minSubtotal} onChange={setMinSubtotal} />
   317→          </div>
   318→          <div className="grid gap-3 sm:grid-cols-2">
   319→            <div className="space-y-1.5">
   320→              <Label htmlFor="c-start">Starts</Label>
   321→              <Input id="c-start" type="datetime-local" value={startsAt} onChange={(e) => setStartsAt(e.target.value)} />
   322→            </div>
   323→            <div className="space-y-1.5">
   324→              <Label htmlFor="c-end">Ends</Label>
   325→              <Input id="c-end" type="datetime-local" value={endsAt} onChange={(e) => setEndsAt(e.target.value)} />
   326→            </div>
   327→          </div>
   328→          <div className="flex items-end gap-4">
   329→            <div className="flex-1 space-y-1.5">
   330→              <Label htmlFor="c-max">Max redemptions (blank = ∞)</Label>
   331→              <Input id="c-max" type="number" min={1} value={maxRedemptions} onChange={(e) => setMaxRedemptions(e.target.value)} />
   332→            </div>
   333→            <label className="flex items-center gap-2 pb-2 text-sm">
   334→              <Checkbox checked={active} onCheckedChange={(c) => setActive(c === true)} /> Active
   335→            </label>
   336→          </div>
   337→        </div>
   338→        <DialogFooter>
   339→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
   340→          <Button onClick={() => void save()} disabled={saving} className="bg-forest text-cream hover:bg-[#23503E]">{saving ? "Saving…" : coupon ? "Save changes" : "Create coupon"}</Button>
   341→        </DialogFooter>
   342→      </DialogContent>
   343→    </Dialog>
   344→  );
   345→}

   353→  const [editing, setEditing] = useState<CampaignAdminDTO | "new" | null>(null);
   354→  const [deleting, setDeleting] = useState<CampaignAdminDTO | null>(null);
   355→
   356→  const load = useCallback(async () => {
   357→    const res = await get<{ campaigns: CampaignAdminDTO[] }>("/api/admin/campaigns");
   358→    if (res) setCampaigns(res.campaigns);
   359→    setLoading(false);
   360→  }, [get]);
   361→
   362→  useEffect(() => {
   363→    let cancelled = false;
   364→    (async () => {
   365→      const res = await get<{ campaigns: CampaignAdminDTO[] }>("/api/admin/campaigns");
   366→      if (!cancelled) {
   367→        if (res) setCampaigns(res.campaigns);
   368→        setLoading(false);
   369→      }
   370→    })();
   371→    return () => {
   372→      cancelled = true;
   373→    };
   374→  }, [get]);
   375→
   376→  const totals = (campaigns ?? []).reduce(
   377→    (acc, c) => ({ impressions: acc.impressions + c.impressions, clicks: acc.clicks + c.clicks, dismissals: acc.dismissals + c.dismissals }),
   378→    { impressions: 0, clicks: 0, dismissals: 0 }
   379→  );
   380→  const ctr = totals.impressions > 0 ? ((totals.clicks / totals.impressions) * 100).toFixed(1) : "0.0";
   381→
   382→  async function setStatus(c: CampaignAdminDTO, status: string) {
   383→    const res = await patchCampaign({ id: c.id, status });

   386→      await load();

   388→  }

   393→      await apiSend(`/api/admin/campaigns?id=${encodeURIComponent(deleting.id)}`, "DELETE");

   395→      setDeleting(null);
   396→      await load();
   397→    } catch (err) {
   398→      toast.error(errMessage(err));
   399→    }
   400→  }
   401→
   402→  return (
   403→    <div className="space-y-4">
   404→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   405→        <Card className="rounded-2xl p-4 shadow-sm"><p className="text-xs uppercase tracking-wide text-muted-foreground">Impressions</p><p className="text-xl font-bold">{totals.impressions.toLocaleString()}</p></Card>
   406→        <Card className="rounded-2xl p-4 shadow-sm"><p className="text-xs uppercase tracking-wide text-muted-foreground">Clicks</p><p className="text-xl font-bold">{totals.clicks.toLocaleString()}</p></Card>
   407→        <Card className="rounded-2xl p-4 shadow-sm"><p className="text-xs uppercase tracking-wide text-muted-foreground">Dismissals</p><p className="text-xl font-bold">{totals.dismissals.toLocaleString()}</p></Card>
   408→        <Card className="rounded-2xl p-4 shadow-sm"><p className="text-xs uppercase tracking-wide text-muted-foreground">Click-through</p><p className="text-xl font-bold text-forest">{ctr}%</p></Card>
   409→      </div>
   410→
   411→      <div className="flex flex-wrap items-center justify-between gap-3">
   412→        <p className="text-sm text-muted-foreground">Popups and banners shown on the storefront.</p>
   413→        <Button onClick={() => setEditing("new")} className="bg-forest text-cream hover:bg-[#23503E]">
   414→          <Plus className="h-4 w-4" /> New campaign
   415→        </Button>
   416→      </div>
   417→
   418→      <div className="rounded-2xl border border-border bg-card shadow-sm">

   420→          <div className="p-4"><SkeletonRows rows={3} /></div>
   421→        ) : (campaigns ?? []).length === 0 ? (
   422→          <div className="p-4"><EmptyState icon={Megaphone} title="No campaigns yet" body="Announce a weekend special to returning customers." /></div>
   423→        ) : (
   424→          <div className="max-h-[56vh] overflow-auto">
   425→            <Table className="min-w-[860px]">
   426→              <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur">
   427→                <TableRow>
   428→                  <TableHead>Campaign</TableHead>
   429→                  <TableHead>Placement</TableHead>
   430→                  <TableHead>Status</TableHead>
   431→                  <TableHead>Schedule</TableHead>
   432→                  <TableHead>Targeting</TableHead>
   433→                  <TableHead className="text-right">Engagement</TableHead>
   434→                  <TableHead className="text-right">Actions</TableHead>
   435→                </TableRow>
   436→              </TableHeader>
   437→              <TableBody>
   438→                {(campaigns ?? []).map((c) => (
   439→                  <TableRow key={c.id}>
   440→                    <TableCell>
   441→                      <p className="max-w-[200px] truncate font-medium">{c.title}</p>

   446→                    <TableCell className="text-xs text-muted-foreground">
   447→                      {c.startsAt ? formatDate(c.startsAt) : "now"} → {c.endsAt ? formatDate(c.endsAt) : "no end"}
   448→                      <span className="block">priority {c.priority}</span>
   449→                    </TableCell>

   451→                    <TableCell className="text-right text-xs">
   452→                      <span className="flex items-center justify-end gap-1"><MousePointerClick className="h-3 w-3" /> {c.clicks}</span>

   454→                    </TableCell>
   455→                    <TableCell className="text-right">
   456→                      <DropdownMenu>
   457→                        <DropdownMenuTrigger asChild>
   458→                          <Button variant="ghost" size="icon" aria-label={`Actions for ${c.title}`}>
   459→                            <MoreHorizontal className="h-4 w-4" />

   491→        onOpenChange={(o) => !o && setDeleting(null)}
   492→        title={deleting ? `Delete “${deleting.title}”?` : "Delete campaign?"}
   493→        description="This removes the campaign and its engagement history."

   509→function toLocalInput(iso: string): string {
   510→  const d = new Date(iso);
   511→  if (Number.isNaN(d.getTime())) return "";
   512→  const pad = (n: number) => String(n).padStart(2, "0");
   513→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   514→}
